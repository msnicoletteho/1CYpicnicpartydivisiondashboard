How to Connect Google Sheets Backend

To receive student submissions and performance data automatically into a Google Sheet:

  1. Create a Google Sheet with columns: Timestamp, StudentName, Class, Score, Level, CommonMistakes.
  2. Go to Extensions > Apps Script, paste the script below, and deploy it as a Web App (execute as 'Me', access as 'Anyone').
  3. Paste your deployed Web App URL into the GOOGLE_APPS_SCRIPT_URL constant at the top of this file.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.Timestamp, data.StudentName, data.Class, data.Score, data.Level, data.CommonMistakes]);
  return ContentService.createTextOutput(JSON.stringify({"status":"success"})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var headers = rows.shift();
  var result = rows.map(r => ({Timestamp:r[0], StudentName:r[1], Class:r[2], Score:r[3], Level:r[4], CommonMistakes:r[5]}));
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
